# Evaluation env ONLY — phase 2 of scripts/setup_vllm_env.sh. Kept apart from
# requirements.txt because vLLM pins its own torch: installing it next to the training stack
# would move the numerics every checkpoint in outputs/ was trained under.
#
# vllm is PINNED, and the pin is about the DRIVER, not about features. This cluster runs
# 570.124.06 = CUDA 12.8, and vLLM's own COMPILED extension (vllm/_C.abi3.so) has to match it.
# Measured by unzipping each wheel and reading the linked runtime out of _C.abi3.so:
#
#     vllm 0.11.0   ->  libcudart.so.12    works here   <- pinned (torch 2.8.0, see phase 1)
#     vllm 0.11.2   ->  libcudart.so.12    also works   (torch 2.9.0)
#     vllm 0.21.0   ->  libcudart.so.13    ImportError: libcudart.so.13
#     vllm 0.26.0   ->  libcudart.so.13    "NVIDIA driver ... too old (found version 12080)"
#     vllm 0.27.1   ->  libcudart.so.13    same
#
# Pinning torch to a cu128 build is NOT sufficient on its own: 0.26.0 pins torch==2.11.0 and
# still ships a CUDA 13 extension, which is why phase 1 and this pin have to move together and
# why the verification imports vllm._C rather than just vllm.
#
# To move this pin forward, re-run that check on the candidate wheel:
#     pip download --no-deps --only-binary=:all: vllm==<v> -d /tmp/w
#     unzip -o /tmp/w/*.whl 'vllm/_C*.so' -d /tmp/w/x
#     strings /tmp/w/x/vllm/_C*.so | grep -o 'libcudart\.so\.[0-9]*' | sort -u
# and update phase 1's torch/torchvision/torchaudio to whatever that release pins.
vllm==0.11.0

# vllm 0.11.0 declares `transformers>=4.55.2` with NO upper bound, so pip installs 5.x, whose
# tokenizer API dropped attributes vLLM still reads. The engine starts, loads the model, and
# then dies mid-generation with
#     AttributeError: LlamaTokenizer has no attribute all_special_tokens_extended
# The cap has to be here because the dependency itself does not carry it.
transformers<5
